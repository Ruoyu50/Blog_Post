```jsx
const express = require("express");
const app = express();

app.get("/", (req, res) => {
  const now = new Date().toLocaleTimeString();
  res.send(`
    <!DOCTYPE html>
    <html>
      <head><title>Dynamic with Node</title></head>
      <body>
        <h1>Welcome!</h1>
        <p>The time on the server is: ${now}</p>
      </body>
    </html>
  `);
});

app.listen(3000, () => {
  console.log("Server running at http://localhost:3000");
});
```

`npm init -y` create a node package

`npm install express` install Express to run Node apps

`node server.js` run file by cmd+right click link
